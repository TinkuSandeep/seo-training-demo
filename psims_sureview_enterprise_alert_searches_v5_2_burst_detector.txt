V5.2 ADDITIONS AND CORRECTIONS
------------------------------
- Adds a production-candidate combined four-host five-minute burst alert.
- Compares the last completed five-minute interval with the identical time slot on the previous five days.
- Uses combined active-active volume to avoid false alerts from traffic redistribution between hosts.
- Counts Event 55 Type=Error separately from informational Event 55 records.
- Corrected verified hostnames to GPVWA/GRVWA (the earlier WMA values were wrong).
- Recreates Today after stats so Day=Today comparisons work.
- Seeds four hosts across today plus five baseline days.
- Requires all five baseline days to contain data before volume alerts fire.
- Normalizes FQDN/lowercase host values to uppercase short hostnames.
- Adds confirmed DMP signal-loss monitoring: EventCode=23 Type=1.
- Adds both warning and critical no-activity searches.

Recommended schedule: every 5 minutes

| table host RawHost Severity Reason DMP_Loss_Events LastSeen SampleMessages

10) WARNING/CRITICAL - COMBINED FOUR-HOST FIVE-MINUTE VOLUME SPIKE
Recommended schedule: every 5 minutes
Recommended throttle: 15 minutes
Trigger: Number of Results > 0
Scope: combined active-active hosts, not per host
------------------------------------------------------------------
index=winos source="WinEventLog:Application" SourceName="CloudLink.Client" earliest=-6d@d latest=now
| eval host=upper(mvindex(split(host,"."),0))
| search host IN ("GPVWA00A0001","GPVWA00A0002","GRVWA00A0001","GRVWA00A0002")
| eval BucketStart=floor(_time/300)*300, CurrentBucket=floor(now()/300)*300-300, CurrentDay=relative_time(CurrentBucket,"@d"), Day=relative_time(BucketStart,"@d"), Slot=BucketStart-Day, CurrentSlot=CurrentBucket-CurrentDay
| where BucketStart=CurrentBucket OR (Day>=relative_time(CurrentDay,"-5d@d") AND Day<CurrentDay AND Slot=CurrentSlot)
| stats count as Total count(eval(EventCode=33)) as E33 count(eval(EventCode=34)) as E34 count(eval(EventCode=55 AND Type="Error")) as E55Error dc(host) as HostsWithData by BucketStart
| eval DataPresent=1
| append [ | makeresults count=6
    | streamstats count as n
    | eval Offset=n-1, CurrentBucket=floor(now()/300)*300-300
    | eval BucketStart=case(Offset=0,CurrentBucket,Offset=1,relative_time(CurrentBucket,"-1d"),Offset=2,relative_time(CurrentBucket,"-2d"),Offset=3,relative_time(CurrentBucket,"-3d"),Offset=4,relative_time(CurrentBucket,"-4d"),true(),relative_time(CurrentBucket,"-5d")), Total=0, E33=0, E34=0, E55Error=0, HostsWithData=0, DataPresent=0
    | fields BucketStart Total E33 E34 E55Error HostsWithData DataPresent ]
| stats sum(Total) as Total sum(E33) as E33 sum(E34) as E34 sum(E55Error) as E55Error max(HostsWithData) as HostsWithData max(DataPresent) as DataPresent by BucketStart
| eval CurrentBucket=floor(now()/300)*300-300, IsCurrent=if(BucketStart=CurrentBucket,1,0)
| eventstats avg(eval(if(IsCurrent=0,Total,null()))) as SameSlotAvg stdev(eval(if(IsCurrent=0,Total,null()))) as SameSlotStdev sum(eval(if(IsCurrent=0,DataPresent,0))) as BaselineDays
| where IsCurrent=1 AND BaselineDays=5 AND SameSlotAvg>=20
| eval PercentAbove=round(((Total-SameSlotAvg)/SameSlotAvg)*100,1), Gap33_34=E33-E34
| where PercentAbove>=20
| eval Severity=if(PercentAbove>=40,"CRITICAL","WARNING"), Reason="Combined CloudLink five-minute volume is above the previous five-day same-slot baseline", WindowStart=strftime(BucketStart,"%Y-%m-%d %H:%M:%S"), Scope="GPVWA00A0001,GPVWA00A0002,GRVWA00A0001,GRVWA00A0002"
| table Severity Reason WindowStart Scope Total SameSlotAvg SameSlotStdev PercentAbove HostsWithData E33 E34 Gap33_34 E55Error BaselineDays
